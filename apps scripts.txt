function doGet(e) {

  var ss = SpreadsheetApp.getActiveSpreadsheet();

  function lerAba(nomeAba) {

    var aba = ss.getSheetByName(nomeAba);

    if (!aba || aba.getLastRow() <= 1) {
      return [];
    }

    var dados = aba.getDataRange().getValues();
    var cabecalho = dados[0];

    var resultado = [];

    for (var i = 1; i < dados.length; i++) {

      var linha = dados[i];
      var obj = {};
      var linhaVazia = true;

      for (var j = 0; j < cabecalho.length; j++) {

        var colunaNome = String(cabecalho[j])
          .trim()
          .toLowerCase();

        var valor = linha[j];

        if (
          valor !== "" &&
          valor !== null &&
          valor !== undefined
        ) {
          linhaVazia = false;
        }

        if (valor instanceof Date) {

          var ano = valor.getFullYear();
          var mes = String(valor.getMonth() + 1).padStart(2, '0');
          var dia = String(valor.getDate()).padStart(2, '0');

          valor = ano + '-' + mes + '-' + dia;
        }

        obj[colunaNome] = valor;
      }

      if (!linhaVazia) {
        resultado.push(obj);
      }
    }

    return resultado;
  }


  var atletas = lerAba("CADASTRO");
  var frequencias = lerAba("FREQUENCIA");
  var atividades = lerAba("ATIVIDADE");
  var desempenhos = lerAba("DESEMPENHO");
  var usuarios = lerAba("USUARIOS"); 
  var desempenhosEscola = lerAba("DESEMPENHO_ESCOLA");


  // ==========================================
  // PRÓXIMO ID DE ATLETA
  // ==========================================

  var maiorIdAtleta = 0;

  atletas.forEach(function(atleta) {

    var id = Number(
      atleta.id_atleta || 0
    );

    if (!isNaN(id) && id > maiorIdAtleta) {
      maiorIdAtleta = id;
    }
  });


  // ==========================================
  // PRÓXIMO ID DE ATIVIDADE
  // ==========================================

  var maiorIdAtividade = 0;

  atividades.forEach(function(atividade) {

    var id = Number(
      atividade.id_atividade || 0
    );

    if (!isNaN(id) && id > maiorIdAtividade) {
      maiorIdAtividade = id;
    }
  });


  // ==========================================
  // PRÓXIMO ID DE DESEMPENHO
  // ==========================================

  var maiorIdDesempenho = 0;

  desempenhos.forEach(function(desempenho) {

    var id = Number(
      desempenho.id_desempenho || 0
    );

    if (!isNaN(id) && id > maiorIdDesempenho) {
      maiorIdDesempenho = id;
    }
  });


  // ==========================================
  // RESUMO DE FREQUÊNCIA
  // ==========================================

  var resumoPresencas = {};

  frequencias.forEach(function(freq) {

    var id = String(
      freq.id_atleta || ''
    ).trim();

    var situacao = String(
      freq.situacao || ''
    ).trim().toUpperCase();

    if (!id) {
      return;
    }

    if (!resumoPresencas[id]) {

      resumoPresencas[id] = {
        presencas: 0,
        faltas: 0,
        total: 0
      };

    }

    resumoPresencas[id].total++;

    if (situacao === 'P') {
      resumoPresencas[id].presencas++;
    }

    if (situacao === 'F') {
      resumoPresencas[id].faltas++;
    }

  });


  return ContentService
    .createTextOutput(
      JSON.stringify({

        status: "success",

        timestamp: new Date().toISOString(),

        proximoIdAtleta:
          maiorIdAtleta + 1,

        proximoIdAtividade:
          maiorIdAtividade + 1,

        proximoIdDesempenho:
          maiorIdDesempenho + 1,

        cadastro: atletas,

        frequencia: frequencias,

        atividade: atividades,

        desempenho: desempenhos,

        usuarios: usuarios, 

        desempenho_escola: desempenhosEscola,

        resumo: {
          presencas: resumoPresencas
        }

      }, null, 2)
    )
    .setMimeType(
      ContentService.MimeType.JSON
    );
}



function normalizarMesAnoServidor(valor) {
    const texto = String(valor || '').trim().toLowerCase();

    const meses = {
        jan: '01', janeiro: '01',
        fev: '02', fevereiro: '02',
        mar: '03', marco: '03', março: '03',
        abr: '04', abril: '04',
        mai: '05', maio: '05',
        jun: '06', junho: '06',
        jul: '07', julho: '07',
        ago: '08', agosto: '08',
        set: '09', setembro: '09',
        out: '10', outubro: '10',
        nov: '11', novembro: '11',
        dez: '12', dezembro: '12'
    };

    const partes = texto.split('/');
    if (partes.length !== 2) return '';

    let mes = partes[0].trim();
    let ano = partes[1].trim();

    if (meses[mes]) {
        mes = meses[mes];
    } else if (/^\d{1,2}$/.test(mes)) {
        mes = mes.padStart(2, '0');
    }

    if (!/^\d{4}$/.test(ano) || !/^(0[1-9]|1[0-2])$/.test(mes)) {
        return '';
    }

    return `${mes}/${ano}`;
}



function doPost(e) {

  var lock = LockService.getScriptLock();

  lock.tryLock(10000);

  try {

    var data = JSON.parse(
      e.postData.contents
    );

    var ss =
      SpreadsheetApp.getActiveSpreadsheet();

    var acao = data.acao;

    var aba;


    // ==========================================
    // CADASTRO
    // ==========================================

    if (acao === 'cadastro') {

      aba =
        ss.getSheetByName("CADASTRO");

      aba.appendRow([
        data.id_atleta,
        data.nome,
        data.nascimento,
        data.idade,
        data.responsavel,
        data.contato
      ]);

    }


    // ==========================================
    // FREQUÊNCIA
    // ==========================================

    else if (acao === 'frequencia') {

      aba =
        ss.getSheetByName("FREQUENCIA");

      aba.appendRow([
        data.id_atleta,
        data.nome,
        data.data,
        data.situacao,
        data.justificativa
      ]);

    }


    // ==========================================
    // ATIVIDADE
    // ==========================================

    else if (
      acao === 'atividade' ||
      acao === 'criar_atividade'
    ) {

      aba =
        ss.getSheetByName("ATIVIDADE");

      aba.appendRow([
        data.id_atividade || data.id,
        data.nome_atividade,
        data.data,
        data.situacao,
        data.tipo
      ]);

    }

    // ========================
    //DESEMPENHO ESCOLA 
    // ========================
    
    else if (acao === 'desempenho_escola') {
        aba = ss.getSheetByName("DESEMPENHO_ESCOLA");

        if (!aba) {
            throw new Error('A aba DESEMPENHO_ESCOLA não foi encontrada.');
        }

        const idAtleta = String(data.id_atleta || '').trim();
        const mesAno = normalizarMesAnoServidor(data.mes);

        if (!idAtleta || !mesAno) {
            throw new Error('Atleta e mês/ano são obrigatórios.');
        }

        // A planilha usa:
        // Coluna B = ID do atleta
        // Coluna D = Mês/ano da avaliação
        const ultimaLinha = aba.getLastRow();

        if (ultimaLinha >= 2) {
            const registros = aba.getRange(
                2, 2, ultimaLinha - 1, 3
            ).getDisplayValues();

            const duplicado = registros.some(linha => {
                const idExistente = String(linha[0] || '').trim();
                const mesExistente = normalizarMesAnoServidor(linha[2]);

                return idExistente === idAtleta &&
                      mesExistente === mesAno;
            });

            if (duplicado) {
                return ContentService
                    .createTextOutput(JSON.stringify({
                        result: "error",
                        codigo: "DUPLICADO",
                        error: "Este atleta já possui avaliação escolar neste mês."
                    }))
                    .setMimeType(ContentService.MimeType.JSON);
            }
        }

        aba.appendRow([
            data.id_desp,
            idAtleta,
            data.nome,
            mesAno,
            data.desempenho,
            data.observacao
        ]);
    }





    // ==========================================
    // DESEMPENHO
    // ==========================================

    else if (acao === 'desempenho') {

      aba =
        ss.getSheetByName("DESEMPENHO");

      aba.appendRow([
        data.id_desempenho,
        data.id_atividade,
        data.id_atleta,
        data.nome,
        data.desempenho,
        data.data
      ]);

    }


    else {

      throw new Error(
        "Ação não reconhecida: " + acao
      );

    }


    return ContentService
      .createTextOutput(
        JSON.stringify({
          result: "success"
        })
      )
      .setMimeType(
        ContentService.MimeType.JSON
      );


  } catch (err) {

    return ContentService
      .createTextOutput(
        JSON.stringify({
          result: "error",
          error: err.toString()
        })
      )
      .setMimeType(
        ContentService.MimeType.JSON
      );

  } finally {

    lock.releaseLock();

  }
}
